DECLARE 
	@CUST NVARCHAR(5) = 'ZZZCustZZZ',			--- !!! Customer Account Code (3-5 chars) OR alphabetically by first letter: a b c e g h l m n o p q r s t v w y
		@execSectbAccess INT = ZZZexecSectbAccessZZZ,		--1 to insert/UPDATE Sectb_access, 0 to NOT  (Does NOT include LogicMonitor)
		@DOMAIN NVARCHAR(30) = 'ASPGOV',
		@EXECUTENOW INT = 0,					--1 to execute now, 0 to just print)

		@LNAME NVARCHAR(30),		--Last Name
		@EMAILADDY NVARCHAR(100),	--Email Address
		@SECTBUID NVARCHAR(8),		--sectb tables UID (user, crosswalk AND access) LEAVE =NULL TO SET DYNAMICALLY, OR EDIT TO MANUALLY OVERRIDE ... UID/SPIUSER has a max of 8 chars, the crosswalk table matches it to the full AD domain account, so this @SECTBUID can be anything AS long AS its unique... also make sure there are no duplicates
		@SAMID NVARCHAR(30),		--AD SamAccountName
		@ISCUSTADMIN CHAR(1) = 'N',				--user IS an Admin in Customer's App Environment?

		@CurrentDate NVARCHAR(19) = FORMAT(GETDATE(),'yyyy-MM-dd 00:00:00'),
		@CurrentTime NVARCHAR(19) = FORMAT(GETDATE(),'HH:mm:ss'),

		@SQLCMD_Login NVARCHAR(max),
		@SQLCMD_DB NVARCHAR(max),
		@SQLCMD_DBUser NVARCHAR(max),
		@SQLCMD_UsrViw NVARCHAR(max),
		@SQLCMD_DBViews NVARCHAR(max),
		@SQLCMDcom NVARCHAR(max),
		@SQLCMDnotattach NVARCHAR(max),
		@SQLCMD_sectb_access NVARCHAR(max),
		@SQLCMDEND NVARCHAR(max),
		@SQLCMDadmin NVARCHAR(max),
		@SQLCMDlogicmon NVARCHAR(max),
		@SQLCMD_FinPro NVARCHAR(max),
		@SectbAccessMin_EFP NVARCHAR(4) = '550',		--sectb_access minimum resource code to get to Main Menu, FinancePLUS
		@SectbAccessMin_ECP NVARCHAR(4) = '4901',		--sectb_access minimum resource code to get to Main Menu, CommunityPLUS
		@DB_Name NVARCHAR(100),
		@52sqllogin NVARCHAR(60) = 'nunya',

--per user begins here

	WHILE @@FETCH_STATUS = 0
		BEGIN
			--extract samaccountname FROM AD login
			SET @SAMID =REPLACE(@DOMID,'ASPGOV\','')

			IF (@SECTBUID IS NULL)
				BEGIN
					SET @SECTBUID = LOWER(@SAMID)
					UPDATE @tbl_USERS SET sectbuid = @SECTBUID WHERE loginid = @DOMID
				END

			SET @FNAME = UPPER(@FNAME)
			SET @LNAME = UPPER(@LNAME)

			PRINT 'PRINT ''--- BEGIN user: ' + @FNAME + ' ' + @LNAME + ' = [' + @DOMID + '] (' + @SECTBUID + ') ----------------------------------------------------------------------------------------'' ; SET NOCOUNT ON
			'

			IF (@ISCUSTADMIN = 'Y') 
				BEGIN 
					SET @SQLCMDadmin = 'EXEC sp_addrolemember ''PLUS_admin'',''' + @SAMID + '''
	EXEC sp_addrolemember ''PLUS_datawriter'',''' + @SAMID + '''
	EXEC sp_addrolemember ''PLUS_datareader'',''' + @SAMID + '''
	GO'
				END
			ELSE
				BEGIN 
					SET @SQLCMDadmin = 'EXEC sp_droprolemember ''PLUS_admin'',''' + @SAMID + '''
	EXEC sp_addrolemember ''PLUS_datawriter'',''' + @SAMID + '''
	EXEC sp_addrolemember ''PLUS_datareader'',''' + @SAMID + '''
	GO'
				END
	
			
			--@SQLCMD_Login is for all DBs
			SET @SQLCMD_Login = 
	'USE [master]
	IF NOT EXISTS (SELECT name FROM syslogins WHERE name = '''+@DOMID+''')
		BEGIN
			PRINT ''Creating SQL Login ' + quotename(@DOMID) + ' on ' + @@SERVERNAME + '''
			CREATE LOGIN ' + quotename(@DOMID) + ' FROM WINDOWS WITH DEFAULT_DATABASE=[master]
		END
	ELSE
		BEGIN
			PRINT ''SQL Login ' + quotename(@DOMID) + ' already exists on ' + @@SERVERNAME + '''
		END
--------------------------------------------------------------------------------------------------------------
'
PRINT @SQLCMD_Login

	BEGIN
		EXEC (@SQLCMD_Login)
	END

			SET @SQLCMD_DB =

GO
PRINT ''--------BEGIN ' + @SAMID + ' on [' + @DB_Name + ']---------------------------------------------------'''
			SET @SQLCMD_DBUser = '

			EXEC sp_adduser '''+ @DOMID + ''',''' + @SAMID + '''

		END

			IF NOT EXISTS (SELECT * FROM sys.schemas WHERE name = ''' + @SAMID + ''')
				BEGIN

				END
			PRINT ''User ' + quotename(@SAMID) + ' already exists in ' + quotename(@DB_Name) + ' for Login ' + quotename(@DOMID) + '''
			--ALTER USER ' + quotename(@SAMID) + ' WITH LOGIN = ' + quotename(@DOMID) + '
		END
	' + @SQLCMDadmin + '
'
			SET @SQLCMDnotattach = '

	INSERT INTO [' + @DB_Name + '].[dbo].[sectb_user] (uid,lname,fname,email,bld_group,dept,user_dba,qwarn,qlimit,stampid,stampdate,stamptime) VALUES(''' + @SECTBUID + ''',''' + @LNAME + ''',''' + @FNAME + ''',''' + @EMAILADDY + ''',''>=0'',''*'',''' + @ISCUSTADMIN + ''',''0'',''0'',''' + LEFT(@CurrentUser,8) + ''',''' + @CurrentDate + ''',''' + @CurrentTime + ''')
	PRINT ''Added ' + @SECTBUID + ' to sectb_user''

	-- SECTB_CROSSWALK --------------------
	IF EXISTS (SELECT * FROM [' + @DB_Name + '].[dbo].[sectb_crosswalk] WHERE spiuser = ''' + @SECTBUID + ''' OR winuser = ''' + @SAMID + ''') BEGIN DELETE FROM [' + @DB_Name + '].[dbo].[sectb_crosswalk] WHERE spiuser = ''' + @SECTBUID + ''' OR winuser = ''' + @SAMID + ''' END
	INSERT INTO [' + @DB_Name + '].[dbo].[sectb_crosswalk](spiuser,windomain,winuser) VALUES(''' + @SECTBUID + ''',''' + @DOMAIN + ''',''' + @SAMID + ''')
	PRINT ''Added ' + @SECTBUID + ' to sectb_crosswalk'''

			SET @SQLCMD_sectb_access = '
	-- SECTB_ACCESS --------------------
	IF EXISTS (SELECT * FROM [' + @DB_Name + '].[dbo].[sectb_resource])

			INSERT INTO [' + @DB_Name + '].[dbo].[sectb_access]
			 SELECT ''' + @SECTBUID + ''',usertype,''' + @CurrentUser + ''',''' + @CurrentDate + ''',''' + @CurrentTime + ''' FROM [' + @DB_Name + '].[dbo].[sectb_resource]

			PRINT ''Added ' + @SECTBUID + ' to sectb_access''
		END 
	ELSE 
		BEGIN 
			PRINT ''No resource codes yet - cannot add to sectb_access''
		END'

			SET @SQLCMD_UsrViw = '
	-- SECTB_USRVIW --------------------
	IF NOT EXISTS (SELECT * FROM ['+@DB_Name+'].[dbo].[sectb_usrviw] WHERE user_id='''+@SECTBUID+''')
		BEGIN
			INSERT INTO ['+@DB_Name+'].[dbo].[sectb_usrviw]
			SELECT distinct ''' + @SECTBUID + ''',viewcode,'' 1=1'','''+@CurrentDate+''',''' + @CurrentUser + ''','''+@CurrentDate+''',''' + @CurrentUser + ''' FROM sectb_usrviw
			WHERE viewcode in (''ACCT'',''BUDACCT'',''BUDORGN'',''BUDPROJ'',''ORGN'',''PROJ'')
		END
	ELSE
		BEGIN
			PRINT ''' + @SECTBUID + ' already exists in ['+@DB_Name+'].[dbo].[sectb_usrviw] ''
		END
	'
			
			SET @SQLCMD_DBViews = '
	-- DB VIEWS --------------------
	IF NOT (SELECT COUNT (*) FROM [' + @DB_Name + '].[sys].[views] v INNER JOIN [' + @DB_Name + '].[sys].[schemas] s ON v.schema_id = s.schema_id WHERE s.name = ''' + @SAMID + ''') = 6
		BEGIN
			EXEC (''CREATE VIEW [' + @SAMID + '].[account] AS SELECT * FROM [faaccount] WHERE 1=1'')
			EXEC (''CREATE VIEW [' + @SAMID + '].[budacct] AS SELECT * FROM [bubudacct] WHERE 1=1'')
			EXEC (''CREATE VIEW [' + @SAMID + '].[budorgn] AS SELECT * FROM [bubudorgn] WHERE 1=1'')
			EXEC (''CREATE VIEW [' + @SAMID + '].[budproj] AS SELECT * FROM [bubudproj] WHERE 1=1'')
			EXEC (''CREATE VIEW [' + @SAMID + '].[orgn] AS SELECT * FROM [faorgn] WHERE 1=1'')
			EXEC (''CREATE VIEW [' + @SAMID + '].[project] AS SELECT * FROM [faproject] WHERE 1=1'')
			PRINT ''SUCCESS: Views Created FOR ' + @SAMID + ' in ['+@DB_Name+']''
		END
	ELSE
		BEGIN
			PRINT ''6 Views already exist FOR ' + @SAMID + ' in ['+@DB_Name+']''
		END
	'
	
			SET @SQLCMD_FinPro = '
	-- spi_CheckSecurityByResource --------------------
	GRANT EXECUTE ON [dbo].[spi_CheckSecurityByResource] TO [PLUS_datareader] ;
	
	-- 52 MIGRATE LOGIN --------------------
	IF (@@SERVERNAME LIKE ''%SPLSDB004%'')
		BEGIN
			IF (DB_NAME() LIKE ''%trn%'')
				BEGIN
					EXEC [efpcatalog_SAP201].[dbo].[dba_MigrateLoginsTo52] ''' + @DB_Name + ''',''' + left(@DB_Name,3) + ''',''' + @SAMID + '''
				END
			ELSE
				BEGIN
					EXEC [efpcatalog_SAP202].[dbo].[dba_MigrateLoginsTo52] ''' + @DB_Name + ''',''' + left(@DB_Name,3) + ''',''' + @SAMID + '''
				END
		END
	ELSE
		BEGIN
			EXEC [efpcatalog].[dbo].[dba_MigrateLoginsTo52] ''' + @DB_Name + ''',''' + left(@DB_Name,3) + ''',''' + @SAMID + '''
		END
	'
	
			SET @SQLCMDcom = 
	'
	-- COMPLUS --------------------
	BEGIN
		EXEC sp_addrolemember ''db_ddladmin'',''' + @SAMID + '''
	END
	'
			
			SET @SQLCMDlogicmon = '
	-- LogicMonitor Login Access -----------
	BEGIN
		EXEC sp_addrolemember ''PLUS_datareader'',''' + @SAMID + '''
		EXEC sp_addrolemember ''PLUS_datawriter'',''' + @SAMID + '''

		IF NOT EXISTS (SELECT * FROM ['+@DB_Name+'].[dbo].[sectb_access] WHERE uid='''+@SECTBUID+''')
			BEGIN
				INSERT INTO ['+@DB_Name+'].[dbo].[sectb_access]
				 VALUES( ''' + @SECTBUID + ''',''@SectbAccessMin@'',''' + @CurrentUser + ''',''' + @CurrentDate + ''',''' + @CurrentTime + ''')
			END
	END
	'
			
			SET @SQLCMDEND = '
	PRINT ''--------FINISHED ' + @SAMID + ' ON [' + @DB_Name + ']---------------------------------------------------''
			GO
------------------------------------------------------------------------------------------------------------------------------------------------'
		IF @DB_Name LIKE '%attach%' SET @SQLCMD_DBUser =REPLACE(REPLACE(@SQLCMD_DBUser,'PLUS_admin','db_datareader'),''',''',''',''ASPGOV\')
		IF @DB_Name LIKE '%attach%' SET @SQLCMD_DBUser = ''       /* If adding domain groups to db_datareader doesn't work, remove this line */
		PRINT @SQLCMD_DB
			IF @EXECUTENOW = 1
				BEGIN
					EXEC (@SQLCMD_DB)
				END
		PRINT @SQLCMD_DBUser
			IF @EXECUTENOW = 1
				BEGIN
					EXEC (@SQLCMD_DBUser)
				END
		IF @DB_Name NOT LIKE '%attach%'
			BEGIN
				PRINT @SQLCMDnotattach
					IF @EXECUTENOW = 1
						BEGIN
							EXEC (@SQLCMDnotattach)
						END

							BEGIN
								EXEC (@SQLCMD_sectb_access)
							END
					END
			END

					IF @EXECUTENOW = 1
						BEGIN
							EXEC (@SQLCMD_UsrViw)
							EXEC (@SQLCMD_UsrViw)
						END
				IF @SAMID LIKE 'logicmon%'
					BEGIN
						SET @SQLCMDlogicmon =REPLACE(@SQLCMDlogicmon,'@SectbAccessMin@',@SectbAccessMin_EFP)
						PRINT @SQLCMDlogicmon
					END
				IF (@DB_Name LIKE '%finpro%') PRINT @SQLCMD_FinPro
			END
		IF @DB_Name LIKE '%complus%' OR @DB_Name LIKE '%compro%'
			BEGIN
				PRINT @SQLCMDcom
					IF @EXECUTENOW = 1
						BEGIN
							EXEC (@SQLCMDcom)
						END
				IF @SAMID LIKE 'logicmon%'
					BEGIN
						SET @SQLCMDlogicmon =REPLACE(@SQLCMDlogicmon,'@SectbAccessMin@',@SectbAccessMin_ECP)
						PRINT @SQLCMDlogicmon
					END
			END
		PRINT @SQLCMDEND
			IF @EXECUTENOW = 1
				BEGIN
					EXEC (@SQLCMDEND)
				END
		
		SET @cursor_count = @cursor_count+1

		FETCH NEXT FROM db_cursor INTO @DB_Name
	END
CLOSE db_cursor
